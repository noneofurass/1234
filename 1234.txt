def get_book_details(book_url):

    try:
        response = session.get(book_url, timeout=10)
        soup = BeautifulSoup(response.text, "html.parser")

        table = soup.find("table", class_="table table-striped")

        product_info = {}
        for row in table.find_all("tr"):
            product_info[row.th.text] = row.td.text

        description = ""
        desc = soup.find("div", id="product_description")
        if desc:
            description = desc.find_next("p").text

        category = soup.find("ul", class_="breadcrumb").find_all("li")[2].text.strip()

        return {
            "UPC": product_info.get("UPC"),
            "Product Type": product_info.get("Product Type"),
            "Price (excl. tax)": product_info.get("Price (excl. tax)"),
            "Price (incl. tax)": product_info.get("Price (incl. tax)"),
            "Tax": product_info.get("Tax"),
            "Availability": product_info.get("Availability"),
            "Number of Reviews": product_info.get("Number of reviews"),
            "Category": category,
            "Description": description
        }

    except:
        return {}


from concurrent.futures import ThreadPoolExecutor

page = 1

while True:

    url = BASE_URL + f"catalogue/page-{page}.html"

    response = session.get(url)

    if response.status_code != 200:
        break

    soup = BeautifulSoup(response.text, "html.parser")

    articles = soup.find_all("article", class_="product_pod")

    page_books = []

    for article in articles:

        product_link = article.h3.a["href"].replace("../../../", "")
        product_url = CATALOGUE_URL + product_link

        page_books.append({
            "Title": article.h3.a["title"],
            "Price": article.find("p", class_="price_color").text,
            "Rating": article.p["class"][1],
            "Availability": article.find("p", class_="instock availability").text.strip(),
            "Product URL": product_url
        })

    with ThreadPoolExecutor(max_workers=50) as executor:
        details = list(executor.map(lambda x: get_book_details(x["Product URL"]), page_books))

    for basic, detail in zip(page_books, details):
        books.append({**basic, **detail})

    print(f"Page {page} completed | Books: {len(books)}")

    page += 1

print("Total Books:", len(books))

def scrape_indian_dishes():
    
    urls = [
        ("https://en.wikipedia.org/wiki/List_of_Indian_dishes", "General"),
        ("https://en.wikipedia.org/wiki/North_Indian_cuisine", "North"),
        ("https://en.wikipedia.org/wiki/South_Indian_cuisine", "South"),
        ("https://en.wikipedia.org/wiki/East_Indian_cuisine", "East"),
        ("https://en.wikipedia.org/wiki/West_Indian_cuisine", "West")
    ]
    
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
    }
    
    scraped_data = []
    recipe_id_counter = 1
    
    gi_tagged_foods = [
        "rasgulla", "hyderabadi haleem", "rogan josh", "chhena poda", 
        "masor tenga", "darjeeling tea", "kanchipuram silk", "bardhaman sitabhog",
        "palani panchamirtham", "tirupati laddu", "mucha kora"
    ]

    print("Initiating Web Scraping from live web sources...")
    
    for url, default_region in urls:
        try:
            res = requests.get(url, headers=headers, timeout=12)
            if res.status_code != 200:
                continue
                
            soup = BeautifulSoup(res.text, 'html.parser')
            current_region = default_region
            
            for element in soup.find_all(['h2', 'h3', 'table']):
                
                if element.name in ['h2', 'h3']:
                    hdr = element.get_text().strip()
                    for r in ['North', 'South', 'East', 'West', 'Central', 'North-East', 'Northeast']:
                        if r.lower() in hdr.lower():
                            current_region = "North-East" if ("east" in r.lower() and "north" in r.lower()) else r.capitalize()
                            break
                            
                elif element.name == 'table' and 'wikitable' in element.get('class', []):
                    rows = element.find_all('tr')
                    for row in rows[1:]:
                        cols = row.find_all(['td', 'th'])
                        if len(cols) < 2:
                            continue
                        
                        col_texts = [c.get_text().strip() for c in cols]
                        name = re.sub(r'\[.*?\]', '', col_texts[0]).strip()
                        
                        if not name or name.lower() in ['name', 'dish', 'image', 'main ingredient']:
                            continue
                            
                        a_tag = cols[0].find('a')
                        recipe_url = f"https://en.wikipedia.org{a_tag['href']}" if (a_tag and a_tag.has_attr('href')) else url
                        
                        img_tag = row.find('img')
                        image_url = f"https:{img_tag['src']}" if (img_tag and img_tag.has_attr('src')) else "https://images.india.org/recipes/default.jpg"
                        
                        desc = col_texts[1] if len(col_texts) > 1 else ""
                        if len(col_texts) > 2 and ("ingredient" in col_texts[2].lower() or "veg" in col_texts[2].lower()):
                            desc += " " + col_texts[2]
                        
                        full_text = f"{name} {desc}".lower()
                        
                        diet = "Non-Vegetarian" if any(w in full_text for w in ["chicken", "mutton", "fish", "pork", "meat", "egg", "prawn", "non-veg"]) else "Vegetarian"
                        
                        states = ["Punjab", "Tamil Nadu", "Karnataka", "West Bengal", "Gujarat", "Maharashtra", 
                                  "Kerala", "Rajasthan", "Assam", "Nagaland", "Odisha", "Bihar", "Madhya Pradesh", 
                                  "Jammu and Kashmir", "Telangana", "Andhra Pradesh", "Goa", "Sikkim", "Manipur"]
                        extracted_state = "General"
                        for st in states:
                            if st.lower() in full_text:
                                extracted_state = st
                                break
                        if extracted_state == "General":
                            region_state_map = {"North": "Punjab", "South": "Tamil Nadu", "East": "West Bengal", "West": "Maharashtra", "Central": "Madhya Pradesh", "North-East": "Assam"}
                            extracted_state = region_state_map.get(current_region, "Delhi")
                        
                        methods = ["Deep Frying", "Steaming", "Boiling", "Slow Cooking", "Roasting", "Baking", "Simmering"]
                        method = "Simmering"
                        for m in methods:
                            if m.split()[0].lower() in full_text:
                                method = m
                                break
                                
                        category = "Lunch"
                        if any(w in full_text for w in ["sweet", "dessert", "sugar", "syrup"]):
                            category = "Dessert"
                        elif any(w in full_text for w in ["snack", "street food", "chaat", "fried"]):
                            category = "Snack"
                        elif any(w in full_text for w in ["breakfast", "dosa", "idli", "paratha", "poha"]):
                            category = "Breakfast"
                        elif any(w in full_text for w in ["curry", "biryani", "gravy"]):
                            category = "Dinner" if random.random() > 0.5 else "Lunch"

                        recipe_entry = {
                            "recipe_id": f"SCRAPED_{recipe_id_counter:04d}",
                            "recipe_name": name,
                            "state": extracted_state,
                            "region": current_region if current_region != "General" else "North",
                            "category": category,
                            "diet_type": diet,
                            "ingredients": desc if desc else "Spices, Vegetables, Herbs, Oil",
                            "cooking_method": method,
                            "prep_time_min": random.choice([15, 20, 25, 30, 40, 45, 60, 90]),
                            "recommended_meal_time": "Morning" if category == "Breakfast" else ("Evening" if category in ["Snack", "Dessert"] else "Afternoon/Night"),
                            "gi_tag": "Yes" if any(gi in name.lower() for gi in gi_tagged_foods) else "No",
                            "calories_kcal": random.randint(180, 650),
                            "recipe_url": recipe_url,
                            "image_url": image_url
                        }
                        scraped_data.append(recipe_entry)
                        recipe_id_counter += 1
                        
        except Exception as e:
            print(f"Scraping notice for {url}: {e}")

    return scraped_data

def clean_ingredients_text(ing):
    if not isinstance(ing, str):
        return []
    
    words = re.sub(r'[^a-zA-Z\s,]', '', ing).lower().split(',')
    cleaned = [w.strip() for w in words if len(w.strip()) > 2]
    return cleaned if cleaned else ["spices", "herbs"]

df["cleaned_ingredients_list"] = df["ingredients"].apply(clean_ingredients_text)

df["prep_time_min"] = df.groupby("region")["prep_time_min"].transform(lambda x: x.fillna(x.median()))
df["calories_kcal"] = df["calories_kcal"].fillna(df["calories_kcal"].median())

spicy_keywords = ["chili", "chilli", "pepper", "masala", "curry", "mustard", "spicy", "ginger", "garlic"]
def evaluate_spice(ing_list):
    score = sum(1 for item in ing_list if any(sk in item for sk in spicy_keywords))
    if score >= 3:
        return "High"
    elif score >= 1:
        return "Medium"
    return "Low"

df["spice_level"] = df["cleaned_ingredients_list"].apply(evaluate_spice)

def calc_complexity(row):
    if row["prep_time_min"] >= 50 or row["ingredient_count"] >= 6:
        return "Hard"
    elif row["prep_time_min"] >= 25 or row["ingredient_count"] >= 3:
        return "Moderate"
    return "Easy"

df["cooking_complexity"] = df.apply(calc_complexity, axis=1)

df["regional_cuisine_group"] = df["region"].map({
    "North": "Indo-Aryan Northern",
    "South": "Dravidian Southern",
    "East": "Eastern Estuary",
    "West": "Western Coastal & Arid",
    "Central": "Central Heartland",
    "North-East": "Himalayan & Tribal North-East"
}).fillna("Pan-Indian")

def compute_health_score(row):
    score = 70
    if row["cooking_method"] in ["Steaming", "Boiling"]:
        score += 15
    elif row["cooking_method"] == "Deep Frying":
        score -= 20
    if row["diet_type"] == "Vegetarian":
        score += 10
    if row["calories_kcal"] > 500:
        score -= 10
    return max(10, min(100, score))

df["health_score"] = df.apply(compute_health_score, axis=1)

from sklearn.base import clone

def evaluate_models(X_train, X_test, y_train, y_test, models, feature_set_name):

    results = []
    trained_models = {}
    confusion_matrices = {}

    for name, model in models.items():

        # Create a fresh copy of the model
        current_model = clone(model)

        # Training time
        start_time = time.perf_counter()

        current_model.fit(X_train, y_train)

        training_time = time.perf_counter() - start_time

        # Predictions
        y_pred = current_model.predict(X_test)
        y_prob = current_model.predict_proba(X_test)[:, 1]

        # Metrics
        accuracy = accuracy_score(y_test, y_pred)
        precision = precision_score(y_test, y_pred)
        recall = recall_score(y_test, y_pred)
        f1 = f1_score(y_test, y_pred)
        roc_auc = roc_auc_score(y_test, y_prob)

        # Confusion matrix
        cm = confusion_matrix(y_test, y_pred)

        # Store results
        results.append([
            feature_set_name,
            name,
            X_train.shape[1],
            training_time,
            accuracy,
            precision,
            recall,
            f1,
            roc_auc
        ])

        # Store the fresh model
        trained_models[name] = current_model

        # Store confusion matrix
        confusion_matrices[name] = cm

    results_df = pd.DataFrame(
        results,
        columns=[
            "Feature Set",
            "Model",
            "Features/Components",
            "Training Time",
            "Accuracy",
            "Precision",
            "Recall",
            "F1-Score",
            "ROC-AUC"
        ]
    )

    return results_df, trained_models, confusion_matrices

#qno 3

import numpy as np
from scipy.stats import ttest_ind, ttest_rel

# Same dataset
before = [52, 48, 65, 60, 55, 70, 58, 62, 50, 67]
after  = [60, 55, 70, 66, 63, 75, 64, 68, 57, 72]

alpha = 0.05

# --------------------------------------------------
# Independent t-test
# Treats before and after as two independent groups
# --------------------------------------------------

t_ind, p_ind = ttest_ind(before, after)

print("INDEPENDENT T-TEST")
print("------------------")
print("t-statistic:", t_ind)
print("p-value:", p_ind)

if p_ind < alpha:
    print("Significant difference")
else:
    print("No significant difference")


# --------------------------------------------------
# Paired t-test
# Treats before and after as measurements
# from the same students
# --------------------------------------------------

t_pair, p_pair = ttest_rel(before, after)

print("\nPAIRED T-TEST")
print("-------------")
print("t-statistic:", t_pair)
print("p-value:", p_pair)

if p_pair < alpha:
    print("Significant change")
else:
    print("No significant change")
